🐟 Clownfish → Boss + Shield

Paste these blocks into your original game. 🟠 on start blocks go at the top with your setup, 🟢 on game update blocks go anywhere after your sprites are made, and 🔵 edit blocks edit code you already have.

1. Variables & Constants (on start)

🟠 Add to on start 
// Boss + Shield variables
let bossHP = 6
let bossActive = false
let shieldActive = false
let reflectCount = 0
let boss: Sprite = null
let shield: Sprite = null
let bossInterval = 0

2. Spawn Boss at Level 3 (on game update)

🟢 Trigger boss when player reaches Level 3 
game.onUpdate(function () {
    if (info.level() >= 3 && !bossActive) {
        bossActive = true
        spawnBoss()
    }
})

function spawnBoss() {
    boss = sprites.create(assets.image`clownfish_boss`, SpriteKind.Enemy)
    boss.x = 120
    boss.y = 30
    boss.setFlag(SpriteFlag.StayInScreen, true)
    bossInterval = game.runtime()
}

3. Boss Attack — Fire Red Balls

🟢 Boss periodically fires red balls at the player 
game.onUpdate(function () {
    if (bossActive && boss && game.runtime() - bossInterval > 2000) {
        bossInterval = game.runtime()
        let ball = sprites.createProjectileFromSprite(
            assets.image`red_ball`,
            boss,
            0,
            60
        )
        ball.setKind(SpriteKind.Projectile)
    }
})

4. Shield — Tap A to Reflect

🔵 Edit / add A-button handler 
controller.A.onEvent(ControllerButtonEvent.Pressed, function () {
    if (bossActive) {
        // Raise shield briefly
        shieldActive = true
        if (shield == null) {
            shield = sprites.create(assets.image`shield`, SpriteKind.Player)
        }
        shield.setPosition(mySprite.x, mySprite.y - 10)
        pause(400)
        shieldActive = false
        shield.destroy()
        shield = null
    }
})

// Reflect incoming red balls
sprites.onOverlap(SpriteKind.Projectile, SpriteKind.Player, function (projectile, player) {
    if (shieldActive) {
        // Reverse the ball back toward the boss
        projectile.vy = -Math.abs(projectile.vy)
        projectile.setKind(SpriteKind.Friendly)
        reflectCount += 1
        info.setScore(info.score() + 10)
    } else {
        info.changeLifeBy(-1)
        projectile.destroy()
    }
})

5. Reflected Ball Hits Boss

🔵 Edit / add overlap: Friendly vs Enemy 
sprites.onOverlap(SpriteKind.Friendly, SpriteKind.Enemy, function (ball, enemy) {
    ball.destroy()
    bossHP -= 1
    if (bossHP <= 0) {
        boss.destroy(effects.disintegrate, 500)
        boss = null
        bossActive = false
        game.splash("Boss defeated! 🎉")
        info.changeLifeBy(1)
    } else {
        // Brief flash to show hit
        enemy.startEffect(effects.spray, 200)
    }
})

6. Win Condition — 6 Clean Reflects

🟢 Check reflect count each update 
game.onUpdate(function () {
    if (reflectCount >= 6 && bossActive) {
        boss.destroy(effects.disintegrate, 500)
        boss = null
        bossActive = false
        reflectCount = 0
        game.splash("Perfect! 6 reflects — Boss vanquished! 🐟")
        info.changeLifeBy(1)
    }
})
