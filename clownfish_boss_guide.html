4 · Shield control (A)

Press A to raise the bubble for half a second, then a short cooldown. Replace your old left/right handlers with these so facing is remembered.

def raise_shield():
    global is_shielding, shield_ready
    if not shield_ready:
        return
    shield_ready = False
    is_shielding = True
    nemoSprite.set_image(nemoShieldImage)
    music.jump_up.play()

    def lower_shield():
        global is_shielding, shield_ready
        pause(500)                 # shield up 0.5s
        is_shielding = False
        if facing == "right":
            nemoSprite.set_image(myNemoImageFlipped)
        else:
            nemoSprite.set_image(nemoImage)
        pause(400)                 # cooldown
        shield_ready = True
    control.run_in_background(lower_shield)
controller.A.on_event(ControllerButtonEvent.PRESSED, raise_shield)

def on_left_pressed():
    global facing
    facing = "left"
    if not is_shielding:
        nemoSprite.set_image(nemoImage)
controller.left.on_event(ControllerButtonEvent.PRESSED, on_left_pressed)

def on_right_pressed():
    global facing
    facing = "right"
    if not is_shielding:
        nemoSprite.set_image(myNemoImageFlipped)
controller.right.on_event(ControllerButtonEvent.PRESSED, on_right_pressed)

5 · Boss logic

Spawns the boss, draws its health bar, and handles damage. It enrages at half health (faster, double pitches) and ends the game when beaten.

def spawn_boss():
    global boss, boss_health_bar, boss_health_value
    boss = sprites.create(assets.image("shark2"), SpriteKind.boss_enemy)
    boss.set_position(80, 20)
    boss.set_velocity(35, 0)
    boss.set_bounce_on_wall(True)
    boss_health_value = 300
    boss_health_bar = sprites.create(image.create(46, 8), SpriteKind.boss_health_bar)
    update_boss_health_bar()

    def launch_pitch():
        if not boss:
            return
        if boss_health_value > 150:        # phase 1: one homing pitch
            pitch = sprites.create_projectile_from_sprite(enemyBallImage, boss, 0, 0)
            pitch.set_kind(SpriteKind.enemy_projectile)
            pitch.follow(nemoSprite, 45)
            pitch.lifespan = 4000
        else:                              # phase 2: two faster pitches
            for i in range(2):
                pitch = sprites.create_projectile_from_sprite(enemyBallImage, boss, 0, 0)
                pitch.set_kind(SpriteKind.enemy_projectile)
                pitch.follow(nemoSprite, 70)
                pitch.lifespan = 4000
    game.on_update_interval(1500, launch_pitch)

def update_boss_health_bar():
    global boss_health_value, boss_health_bar, boss
    if not boss or not boss_health_bar:
        return
    bar_width = 40
    bar_height = 6
    width_fill = int(bar_width * (boss_health_value / 300))
    img_bar = image.create(bar_width + 2, bar_height + 2)
    img_bar.fill(1)                                  # black border
    img_bar.draw_rect(1, 1, bar_width, bar_height, 2)   # red back
    img_bar.fill_rect(1, 1, width_fill, bar_height, 7)  # green fill
    boss_health_bar.set_image(img_bar)
    boss_health_bar.set_position(boss.x, boss.y - 14)

def update_boss_health():
    global boss_health_value, boss_active, boss
    was_phase_1 = boss_health_value > 150
    boss_health_value -= 50   # damage per reflected pitch
    update_boss_health_bar()
    if was_phase_1 and boss_health_value <= 150:
        boss.set_velocity(60, 0)
        music.play_melody("C C G G A A G -", 150)
        game.show_long_text("The predator is enraged!", DialogLayout.CENTER)
    if boss_health_value <= 0:
        sprites.destroy(boss, effects.fire, 500)
        sprites.destroy(boss_health_bar)
        boss = None
        boss_active = False
        game.show_long_text("You've learned all about clownfish!", DialogLayout.CENTER)
        game.over(True)

def start_boss_fight():
    global boss_active
    boss_active = True
    clear_level_sprites()
    game.show_long_text("A predator boss appears! Press A to raise your shield!", DialogLayout.CENTER)
    spawn_boss()

6 · Reflect overlaps

The heart of it: a pitch hitting nemo while shielding switches teams and flies back at the boss. No shield up = you take the hit.

def on_projectile_hits_player(sprite, pitch):
    if is_shielding:
        pitch.set_kind(SpriteKind.player_projectile)   # switch teams
        pitch.set_image(playerBallImage)               # turn yellow
        if boss:
            pitch.follow(boss, 90)                     # rocket back
        music.power_up.play()
    else:
        sprites.destroy(pitch, effects.spray, 100)
        info.change_life_by(-1)
        music.wawawawaa.play()
        scene.camera_shake(4, 300)
        if info.life() <= 0:
            game.over(False)
sprites.on_overlap(SpriteKind.player, SpriteKind.enemy_projectile, on_projectile_hits_player)

def on_projectile_hits_boss(proj: Sprite, target: Sprite):
    if target == boss:
        sprites.destroy(proj)
        scene.camera_shake(3, 200)
        update_boss_health()
sprites.on_overlap(SpriteKind.player_projectile, SpriteKind.boss_enemy, on_projectile_hits_boss)
